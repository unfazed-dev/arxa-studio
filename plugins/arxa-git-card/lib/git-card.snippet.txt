/** Every read and write goes through the one host route. `arg.sessionId`
 * makes the card SEAT-AWARE (session worktree + branch); a session the org
 * registry does not know falls back to the org primary seat. */
const postAction = async (action, arg) => {
  const r = await fetch(CARD_ROUTE, {
    method: 'POST', headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ action, arg }),
  })
  const b = await r.json().catch(() => ({}))
  if (!b.ok) { const e = new Error(b.error || action); e.code = b.error || action; throw e }
  return b.result
}
const reasonOf = (e) => String((e && (e.code || e.message)) || e || 'unknown').split('\n')[0]
const isNoOrg = (e) => /^(no-org-open|no-workspace|sidebar-not-ready)/.test(reasonOf(e))
const Icon = (name, fallback, size) => {
  const C = P[name] || P[fallback]
  return C ? h(C, size ? { size } : {}) : null
}

function GitDock({ sessionId, postAction: post, notify, t }) {
  const [status, setStatus] = React.useState(null)
  const [pr, setPr] = React.useState(void 0)
  const [absent, setAbsent] = React.useState(false)
  const [collapsed, setCollapsed] = React.useState(true)
  const [editing, setEditing] = React.useState(null)
  const [busy, setBusy] = React.useState(null)
  const [seatSession, setSeatSession] = React.useState(sessionId || null)
  const listId = React.useId()
  const alive = React.useRef(true)
  React.useEffect(() => () => { alive.current = false }, [])

  const seatArg = () => (seatSession ? { sessionId: seatSession } : {})

  const load = React.useCallback(async () => {
    try {
      let r
      try { r = await post('card.status', sessionId ? { sessionId } : {}) }
      catch (e) {
        if (sessionId && /^session-not-found/.test(reasonOf(e))) { r = await post('card.status', {}); if (alive.current) setSeatSession(null) }
        else throw e
      }
      if (!alive.current) return
      setAbsent(false)
      setStatus(r)
    } catch (e) {
      if (!alive.current) return
      if (isNoOrg(e)) { setAbsent(true); return }
      notify('error', t('git.statusFailed', { reason: reasonOf(e) }))
    }
  }, [post, sessionId, notify, t])

  const loadPr = React.useCallback(async () => {
    if (!seatSession) { setPr(null); return }
    try { const r = await post('card.pr.status', { sessionId: seatSession }); if (alive.current) setPr(r) }
    catch (e) { if (alive.current) setPr({ ok: false, reason: reasonOf(e) }) }
  }, [post, seatSession])

  React.useEffect(() => { setSeatSession(sessionId || null); load() }, [sessionId, load])
  React.useEffect(() => {
    const on = () => { load(); if (!collapsed) loadPr() }
    window.addEventListener('arxa-git-card-refresh', on)
    return () => window.removeEventListener('arxa-git-card-refresh', on)
  }, [load, loadPr, collapsed])
  React.useEffect(() => {
    if (!status || collapsed) return
    if (status.seat && status.seat.kind === 'session' && status.linked && !status.localOnly && pr === void 0) loadPr()
  }, [status, collapsed, pr, loadPr])

  if (absent || status === null) return null

  const interactionActive = editing !== null || busy !== null
  const expanded = !collapsed || interactionActive
  const dirty = status.dirty
  const changed = dirty ? dirty.staged + dirty.unstaged + dirty.untracked : null
  const healthKey = status.health === 'ok' ? null : (status.health === 'missing' ? 'git.health.missing' : 'git.health.detached')
  const summary = healthKey ? t(healthKey) : changed === 0 ? t('git.clean') : t('git.changed', { n: changed })
  const branch = (status.seat && status.seat.branch) || 'main'
  const detailParts = []
  if (dirty) {
    if (dirty.staged) detailParts.push(t('git.staged', { n: dirty.staged }))
    if (dirty.unstaged) detailParts.push(t('git.unstaged', { n: dirty.unstaged }))
    if (dirty.untracked) detailParts.push(t('git.untracked', { n: dirty.untracked }))
  }
  if (status.aheadBehind) {
    if (status.aheadBehind.ahead) detailParts.push(t('git.ahead', { n: status.aheadBehind.ahead }))
    if (status.aheadBehind.behind) detailParts.push(t('git.behind', { n: status.aheadBehind.behind }))
  }
  if (status.wipRun) detailParts.push(t('git.wip', { n: status.wipRun }))
  if (status.chip) detailParts.push(String(status.chip))
  const detail = detailParts.length ? detailParts.join(' · ') : summary

  const run = async (key, fn) => {
    setBusy(key)
    try { return await fn() }
    finally { if (alive.current) setBusy((cur) => (cur === key ? null : cur)) }
  }
  const commit = async () => {
    if (editing === null || editing.text.trim() === '') return
    const subject = editing.text.trim()
    await run('commit', async () => {
      try {
        const r = await post('card.commit', { ...seatArg(), subject })
        if (!alive.current) return
        setEditing(null)
        notify(r && r.parked ? 'error' : 'info', r && r.parked ? t('git.parked') : t('git.committed', { subject }))
        load()
      } catch (e) { notify('error', t('git.commitFailed', { reason: reasonOf(e) })) }
    })
  }
  const createPr = async () => {
    const title = editing && editing.kind === 'pr' ? editing.text.trim() : ''
    if (!title) return
    await run('pr', async () => {
      try {
        const p = await post('card.push', seatArg())
        if (p && p.ok === false) throw new Error(p.reason || 'push-failed')
        const r = await post('card.pr.create', { ...seatArg(), title })
        if (!alive.current) return
        setEditing(null)
        notify('info', t('git.pr.created', { n: r && r.pr ? r.pr.number : '?' }))
        loadPr(); load()
      } catch (e) { notify('error', t('git.pr.failed', { reason: reasonOf(e) })) }
    })
  }
  const mergePr = async () => {
    await run('merge', async () => {
      try {
        const r = await post('card.pr.merge', seatArg())
        if (!alive.current) return
        if (r && r.ok === false) throw new Error(r.reason || 'merge-failed')
        notify('info', t('git.pr.merged'))
        loadPr(); load()
      } catch (e) { notify('error', t('git.pr.failed', { reason: reasonOf(e) })) }
    })
  }

  const action = (key, label, icon, opts) => h(P.Tooltip, { key, label, side: 'bottom', delayMs: 500 },
    h('button', {
      type: 'button', className: S.action, 'aria-label': label,
      disabled: Boolean(opts && opts.disabled) || busy !== null,
      onClick: opts && opts.onClick,
    }, icon))
  const editor = (kind, label, onSave) => h('input', {
    autoFocus: true, className: S.editor, 'aria-label': label, value: editing.text,
    onChange: (event) => { setEditing({ kind, text: event.currentTarget.value }) },
    onKeyDown: (event) => {
      if (event.key === 'Escape') { setEditing(null); return }
      if (event.key === 'Enter' && !event.nativeEvent.isComposing) { event.preventDefault(); onSave() }
    },
  })
  const editActions = (onSave) => [
    action('save', t('git.save'), Icon('IconCheckOutline16', 'IconCheckOutline16', 14), { disabled: editing.text.trim() === '', onClick: onSave }),
    action('cancel', t('git.cancel'), Icon('IconCloseOutline16', 'IconCloseFill14', 14), { onClick: () => setEditing(null) }),
  ]

  const rows = []
  // 1. Status
  rows.push(h('li', { key: 'status', className: S.row },
    h('span', { className: S.preview }, detail),
    h('div', { className: S.actions },
      action('refresh', t('git.refresh'), Icon('IconRefreshOutline16', 'IconRefreshOutline16', 14), { onClick: () => { load(); if (pr !== void 0) loadPr() } }))))
  // 2. Commit
  const canCommit = status.health === 'ok' && changed !== null && changed > 0
  const committing = editing !== null && editing.kind === 'commit'
  rows.push(h('li', { key: 'commit', className: S.row },
    committing
      ? editor('commit', t('git.commitSubject'), commit)
      : h('span', { className: S.preview }, canCommit ? summary : t('git.commitHint')),
    h('div', { className: S.actions },
      committing
        ? editActions(commit)
        : action('commit', t('git.commit'), Icon('IconCheckOutline16', 'IconCheckOutline16', 14), { disabled: !canCommit, onClick: () => setEditing({ kind: 'commit', text: '' }) }))))
  // 3. Approve — session seats of a linked org only
  if (status.seat && status.seat.kind === 'session' && status.linked && !status.localOnly) {
    const prOpen = pr && pr.ok && pr.pr
    const prText = pr === void 0 ? t('git.loading')
      : pr && pr.ok === false ? pr.reason
      : prOpen ? t('git.pr.open', { n: pr.pr.number, state: pr.pr.state }) + (pr.checks && pr.checks.state ? ' · ' + t('git.pr.checks', { state: pr.checks.state }) : '')
      : t('git.pr.none')
    const titling = editing !== null && editing.kind === 'pr'
    rows.push(h('li', { key: 'approve', className: S.row },
      titling
        ? editor('pr', t('git.title.subject'), createPr)
        : h('span', { className: S.preview }, prText),
      h('div', { className: S.actions },
        titling
          ? editActions(createPr)
          : [
            action('pr-refresh', t('git.pr.refresh'), Icon('IconRefreshOutline16', 'IconRefreshOutline16', 14), { onClick: loadPr }),
            prOpen
              ? action('merge', t('git.pr.merge'), Icon('IconCheckOutline16', 'IconCheckOutline16', 14), { disabled: !(pr.checks && pr.checks.state === 'green'), onClick: mergePr })
              : action('pr-create', t('git.pr.create'), Icon('IconPlusOutline16', 'IconPlusOutline16', 14), { disabled: pr === void 0, onClick: () => setEditing({ kind: 'pr', text: '' }) }),
          ])))
  }

  return h('div', { className: S.dock, 'data-git-dock': '' },
    h('div', { className: S.panel },
      h('button', {
        type: 'button', className: S.header, 'aria-controls': listId, 'aria-expanded': expanded,
        'aria-label': expanded ? t('git.collapse') : t('git.expand'),
        disabled: interactionActive,
        onClick: () => { setCollapsed((value) => !value) },
      },
        h('span', { className: S.lead, 'aria-hidden': true }, Icon('IconGitBranchOutline14', 'IconBranchOutline16', 14)),
        h('span', { className: S.count }, t('git.head', { branch, summary })),
        h('span', { className: S.chevron, 'aria-hidden': true }, expanded ? Icon('IconChevronDownOutline14', 'IconChevronDownOutline14') : Icon('IconChevronUpOutline14', 'IconChevronUpOutline14'))),
      h('ul', { id: listId, className: S.list, hidden: !expanded }, expanded ? rows : null)))
}
