/** Every read and write goes through the one host route. `arg.sessionId`
 * makes the card SEAT-AWARE (session worktree + branch); a session the org
 * registry does not know falls back to the org primary seat. */
const postAction = async (action, arg) => {
  const r = await fetch(CARD_ROUTE, {
    method: 'POST', headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ action, arg }),
  })
  const b = await r.json().catch(() => ({}))
  if (!b.ok) { const e = new Error(b.error || action); e.code = b.error || action; throw e }
  return b.result
}
const reasonOf = (e) => String((e && (e.code || e.message)) || e || 'unknown').split('\n')[0]
const isNoOrg = (e) => /^(no-org-open|no-workspace|sidebar-not-ready)/.test(reasonOf(e))
const Icon = (name, fallback, size) => {
  const C = P[name] || P[fallback]
  return C ? h(C, size ? { size } : {}) : null
}

function GitDock({ sessionId, postAction: post, notify, t }) {
  const [status, setStatus] = React.useState(null)
  const [pr, setPr] = React.useState(void 0)
  const [absent, setAbsent] = React.useState(false)
  const [collapsed, setCollapsed] = React.useState(true)
  const [editing, setEditing] = React.useState(null)
  const [busy, setBusy] = React.useState(null)
  const [seatSession, setSeatSession] = React.useState(sessionId || null)
  const [mintChip, setMintChip] = React.useState(null)
  // G4: the move waiting on a yes — { key, title, body, ok, run } | null.
  const [confirm, setConfirm] = React.useState(null)
  // G6: the device-flow conversation — null | { starting } | { userCode, verificationUri }.
  const [relink, setRelink] = React.useState(null)
  const listId = React.useId()
  const alive = React.useRef(true)
  // The live device flow, or null: { poll }. A REF, not state — Cancel has to
  // stop the poll synchronously, and a late tick has to be able to tell "my
  // flow" from "the flow the user already dismissed".
  const relinkRun = React.useRef(null)
  React.useEffect(() => () => {
    alive.current = false
    const run = relinkRun.current
    relinkRun.current = null
    if (run) window.clearInterval(run.poll)
  }, [])

  const seatArg = () => (seatSession ? { sessionId: seatSession } : {})

  const load = React.useCallback(async () => {
    try {
      let r
      try { r = await post('card.status', sessionId ? { sessionId } : {}) }
      catch (e) {
        if (sessionId && /^session-not-found/.test(reasonOf(e))) { r = await post('card.status', {}); if (alive.current) setSeatSession(null) }
        else throw e
      }
      if (!alive.current) return
      setAbsent(false)
      setStatus(r)
    } catch (e) {
      if (!alive.current) return
      if (isNoOrg(e)) { setAbsent(true); return }
      notify('error', t('git.statusFailed', { reason: reasonOf(e) }))
    }
  }, [post, sessionId, notify, t])

  const loadPr = React.useCallback(async () => {
    if (!seatSession) { setPr(null); return }
    try { const r = await post('card.pr.status', { sessionId: seatSession }); if (alive.current) setPr(r) }
    catch (e) { if (alive.current) setPr({ ok: false, reason: reasonOf(e) }) }
  }, [post, seatSession])

  React.useEffect(() => { setSeatSession(sessionId || null); setPr(void 0); setMintChip(null); load() }, [sessionId, load])
  React.useEffect(() => {
    const on = () => { load(); if (!collapsed) loadPr() }
    window.addEventListener('arxa-git-card-refresh', on)
    return () => window.removeEventListener('arxa-git-card-refresh', on)
  }, [load, loadPr, collapsed])
  React.useEffect(() => {
    if (!status || collapsed) return
    if (status.seat && status.seat.kind === 'session' && status.linked && !status.localOnly && pr === void 0) loadPr()
  }, [status, collapsed, pr, loadPr])

  // Polling (A3, unchanged from Phase 4): status every 30 s; the PR read only
  // while expanded AND a PR exists — no idle GitHub traffic. Timers live in
  // effects with disposers.
  const hasPr = Boolean(pr && pr.ok && pr.pr)
  React.useEffect(() => {
    const id = window.setInterval(() => { if (busy === null) load() }, 30000)
    return () => window.clearInterval(id)
  }, [load, busy])
  React.useEffect(() => {
    if (collapsed || !hasPr) return
    const id = window.setInterval(() => { if (busy === null) loadPr() }, 30000)
    return () => window.clearInterval(id)
  }, [collapsed, hasPr, loadPr, busy])

  if (absent || status === null) return null

  const interactionActive = editing !== null || busy !== null || confirm !== null || relink !== null
  const expanded = !collapsed || interactionActive
  const dirty = status.dirty
  const changed = dirty ? dirty.staged + dirty.unstaged + dirty.untracked : null
  const healthKey = status.health === 'ok' ? null : (status.health === 'missing' ? 'git.health.missing' : 'git.health.detached')
  const summary = healthKey ? t(healthKey) : changed === 0 ? t('git.clean') : t('git.changed', { n: changed })
  const branch = (status.seat && status.seat.branch) || 'main'
  const detailParts = []
  if (dirty) {
    if (dirty.staged) detailParts.push(t('git.staged', { n: dirty.staged }))
    if (dirty.unstaged) detailParts.push(t('git.unstaged', { n: dirty.unstaged }))
    if (dirty.untracked) detailParts.push(t('git.untracked', { n: dirty.untracked }))
  }
  if (status.aheadBehind) {
    if (status.aheadBehind.ahead) detailParts.push(t('git.ahead', { n: status.aheadBehind.ahead }))
    if (status.aheadBehind.behind) detailParts.push(t('git.behind', { n: status.aheadBehind.behind }))
  }
  // `conflicts === true` only. A null means the probe could not run, and
  // "we did not check" must not render as "there is no conflict".
  if (status.integrate && status.integrate.conflicts === true) detailParts.push(t('git.conflicts'))
  if (status.wipRun) detailParts.push(t('git.wip', { n: status.wipRun }))
  if (status.chip) detailParts.push(String(status.chip))
  // Frame health + runner + main gate ride the same readout (old status
  // slide). A missing part is not a zero — only measured parts are said.
  const frame = status.frame || {}
  const checks = (pr && pr.ok && pr.checks) || null
  const checkState = checks ? (checks.asleep ? 'asleep' : String(checks.state || 'unknown')) : 'none'
  const runnerAsleep = frame.runner === 'asleep' || checkState === 'asleep'
  const sessionSeat = Boolean(status.seat && status.seat.kind === 'session')
  const relinkRequired = Boolean(status.github && status.github.relinkRequired)
  // Leads the detail line: when GitHub has revoked the grant every push, PR
  // and merge on this card fails, and before 2026-09-04 it failed as a bare
  // 401 with nothing naming the one recovery. This is the surface the user is
  // actually standing on when that happens — the sidebar's org modal is not.
  // G6 (2026-09-06): the line says WHAT is wrong; the fix is a button in the
  // status row (relinkAction). A recovery you can only read was the bug.
  if (relinkRequired) detailParts.push(t('git.relinkNeeded'))
  if (status.localOnly) detailParts.push(t('git.localOnly'))
  if (frame.wired) detailParts.push(t(frame.wired === 'ok' ? 'git.frame.wired' : 'git.frame.missing'))
  // host says 'ok' (manifest.frameRunner) or 'online' (live probe) — both are up
  if (frame.runner) detailParts.push(t(frame.runner === 'ok' || frame.runner === 'online' ? 'git.runner.online' : frame.runner === 'asleep' ? 'git.runner.asleep' : 'git.runner.unknown'))
  if (status.main && status.main.checks === 'red') detailParts.push(t('git.mainRed'))
  const detail = detailParts.length ? detailParts.join(' · ') : summary

  const run = async (key, fn) => {
    setBusy(key)
    try { return await fn() }
    finally { if (alive.current) setBusy((cur) => (cur === key ? null : cur)) }
  }
  const commit = async () => {
    if (editing === null || editing.text.trim() === '') return
    const subject = editing.text.trim()
    await run('commit', async () => {
      try {
        const r = await post('card.commit', { ...seatArg(), subject })
        if (!alive.current) return
        setEditing(null)
        notify(r && r.parked ? 'error' : 'info', r && r.parked ? t('git.parked') : t('git.committed', { subject }))
        load()
      } catch (e) { notify('error', t('git.commitFailed', { reason: reasonOf(e) })) }
    })
  }
  const createPr = async () => {
    const title = editing && editing.kind === 'pr' ? editing.text.trim() : ''
    if (!title) return
    await run('pr', async () => {
      try {
        const p = await post('card.push', seatArg())
        if (p && p.ok === false) throw new Error(p.reason || 'push-failed')
        const r = await post('card.pr.create', { ...seatArg(), title })
        if (!alive.current) return
        setEditing(null)
        notify('info', t('git.pr.created', { n: r && r.pr ? r.pr.number : '?' }))
        loadPr(); load()
      } catch (e) { notify('error', t('git.pr.failed', { reason: reasonOf(e) })) }
    })
  }
  const mergePr = async () => {
    await run('merge', async () => {
      try {
        const r = await post('card.pr.merge', seatArg())
        if (!alive.current) return
        if (r && r.ok === false) throw new Error(r.reason || 'merge-failed')
        notify('info', t('git.pr.merged'))
        loadPr(); load()
      } catch (e) { notify('error', t('git.pr.failed', { reason: reasonOf(e) })) }
    })
  }

  /** Q8: run control from the card. `card.pr.status` already surfaced the
   * run ids, so these never guess one — the newest run on the seat's branch
   * is the target. The host refuses a project seat (ciTarget), and a 409 on
   * cancel comes back as `already-finished` rather than an error. */
  const latestRun = pr && pr.ok && Array.isArray(pr.runs) && pr.runs.length > 0 ? pr.runs[0] : null
  const runLive = Boolean(latestRun && latestRun.status !== 'completed')
  const ciControl = (which) => async () => {
    if (!latestRun) return
    await run('ci-' + which, async () => {
      try {
        const r = await post('card.ci.' + which, { ...seatArg(), runId: latestRun.id })
        if (!alive.current) return
        notify('info', r && r.outcome === 'already-finished'
          ? t('git.ci.alreadyFinished')
          : t(which === 'rerun' ? 'git.ci.rerunStarted' : 'git.ci.cancelled'))
        loadPr()
      } catch (e) { notify('error', t('git.ci.failed', { reason: reasonOf(e) })) }
    })
  }
  const rerunCi = ciControl('rerun')
  const cancelCi = ciControl('cancel')
  const openRun = () => { if (latestRun && latestRun.url) window.open(latestRun.url, '_blank', 'noopener') }

  /** D116/B8: a human action — the host never throws; ok:false carries the
   * manual svc.sh instruction. */
  const wake = async () => {
    await run('wake', async () => {
      try {
        const r = await post('card.runner.wake', seatArg())
        if (!alive.current) return
        notify(r && r.ok ? 'info' : 'error', r && r.ok ? t('git.wakeStarted') : t('git.wakeManual', { reason: String((r && r.reason) || '') }))
        load(); if (pr !== void 0) loadPr()
      } catch (e) { notify('error', t('git.wakeManual', { reason: reasonOf(e) })) }
    })
  }
  /** Stage boundary: mint a version on the merged seat; the chip it returns
   * joins the approve readout until the session changes. */
  const mint = async () => {
    await run('mint', async () => {
      try {
        const r = await post('version.mint', seatArg())
        if (!alive.current) return
        const chip = r && r.chip ? r.chip : null
        if (chip) setMintChip(chip)
        notify('info', t('git.minted', { chip: chip ? (chip.label || String(chip)) : '' }))
        load()
      } catch (e) { notify('error', t('git.mintFailed', { reason: reasonOf(e) })) }
    })
  }
  /** Q6: the engine never drafts — the host returns EVIDENCE and the session
   * model writes the subject. Prefill the composer through the native value
   * setter so React sees the input. */
  const askDraft = async () => {
    await run('draft', async () => {
      try {
        const r = await post('card.commit.draft', seatArg())
        if (!alive.current) return
        const ta = document.querySelector("[data-slot='conversation.composer.bar'] textarea")
        if (!ta) throw new Error('composer-not-found')
        const text = 'Draft a conventional commit subject for the current changes. ' + String((r && r.rule) || '')
          + ' Evidence — stat: ' + String((r && r.uncommittedStat) || '(committed via WIP)')
          + ' Recent stage subjects: ' + (((r && r.recentStageSubjects) || []).join(' | '))
        const setter = Object.getOwnPropertyDescriptor(HTMLTextAreaElement.prototype, 'value')
        if (setter && setter.set) setter.set.call(ta, text); else ta.value = text
        ta.dispatchEvent(new Event('input', { bubbles: true }))
        ta.focus()
        window.__arxaCardDraftRule = r && r.rule
      } catch (e) { notify('error', t('git.draftFailed', { reason: reasonOf(e) })) }
    })
  }
  /** G4 (grilled 2026-09-06): the four moves that are hard to walk back —
   * merge, integrate, finish, mint — set this instead of acting, and the one
   * stock Modal below asks. Same grammar as the sidebar's Delete and the
   * Claude row's Sign out; the copy per key lives in the dictionaries. */
  const ask = (key, go) => () => setConfirm({
    key,
    title: t('git.confirm.' + key + '.title'),
    body: t('git.confirm.' + key + '.body'),
    ok: t('git.confirm.' + key + '.ok'),
    go,
  })
  const confirmGo = () => { const c = confirm; setConfirm(null); if (c) c.go() }
  /** D113: end of life. finishSession refuses unless the branch is merged AND
   * the worktree is clean, and card.status already asked it (dryRun) for the
   * enabled state — so this just runs the same function for real. Behind the
   * same G4 confirm as the other moves that are hard to walk back: the worktree
   * and the branch are both gone afterwards. */
  const finishSession = async () => {
    await run('finish', async () => {
      try {
        await post('card.finish', seatArg())
        if (!alive.current) return
        notify('info', t('git.finish.done'))
        load()
      } catch (e) { notify('error', t('git.finishFailed', { reason: reasonOf(e) })) }
    })
  }
  /** G6: the device flow, the sidebar's relink verbatim — card.github.link
   * LONG-POLLS until GitHub confirms, while card.github.device is polled
   * meanwhile for the one-time code to show. Both ride the card's own route:
   * its host already holds the same getGithub the sidebar does, so the card
   * keeps its one-wire property instead of reaching across. */
  /** Cancel: stop the poll, drop the run token, close. Shipped 2026-09-06
   * without this and the dialog could not be closed — the poll below was
   * cleared only when card.github.link SETTLED, and that long-polls until
   * GitHub confirms, so Cancel closed the modal for one tick and the next
   * 700ms tick reopened it. The `if` guard in startRelink then swallowed
   * every further click. Dropping the token is what makes late writers
   * no-ops; clearing the interval is what stops them happening at all.
   *
   * What this does NOT do is stop the flow on GitHub's side: the host is
   * still polling, and there is no cancel in the device grant. That flow
   * dies on its own when the device code expires (900s, per GitHub's device
   * flow docs). The host-side dedupe is what keeps a re-click from racing
   * it — see card.github.link. */
  const stopRelink = () => {
    const run = relinkRun.current
    relinkRun.current = null
    if (run) window.clearInterval(run.poll)
    setRelink(null)
  }
  const startRelink = () => {
    if (relinkRun.current !== null) return
    const run = { poll: 0 }
    relinkRun.current = run
    setRelink({ starting: true })
    run.poll = window.setInterval(() => {
      post('card.github.device').then((d) => {
        if (alive.current && relinkRun.current === run && d && d.userCode) setRelink(d)
      }, () => {})
    }, 700)
    post('card.github.link').then(() => {
      window.clearInterval(run.poll)
      const mine = relinkRun.current === run
      if (mine) relinkRun.current = null
      if (!alive.current) return
      // The grant really did change, so the card reloads either way; only the
      // dialog and the toast belong to whoever was still watching.
      if (mine) { setRelink(null); notify('info', t('git.relink.done')) }
      load(); if (pr !== void 0) loadPr()
    }, (e) => {
      window.clearInterval(run.poll)
      const mine = relinkRun.current === run
      if (mine) relinkRun.current = null
      if (!alive.current || !mine) return
      setRelink(null)
      notify('error', t('git.relink.failed', { reason: reasonOf(e) }))
    })
  }
  /** Insight links: right-panel surfaces owned by artifact-viewer (A3). */
  const openInsight = (view) => {
    try { window.dispatchEvent(new CustomEvent('arxa-av-open', { detail: { kind: 'insight', view, sessionId: seatSession } })) } catch { /* no bridge */ }
  }
  const wakeAction = () => action('wake', t('git.wake'), Icon('IconPlayOutline16', 'IconTriangleRightFill14', 14), { onClick: wake })
  // A thunk, like wakeAction: `action` is declared BELOW, so an eager const
  // here would hit the TDZ and blank the whole card.
  const relinkAction = () => action('relink', t('git.relink'), Icon('IconLinkOutline16', 'IconLinkOutline14', 14), { onClick: startRelink })

  const action = (key, label, icon, opts) => h(P.Tooltip, { key, label, side: 'bottom', delayMs: 500 },
    h('button', {
      type: 'button', className: S.action, 'aria-label': label,
      disabled: Boolean(opts && opts.disabled) || busy !== null,
      onClick: opts && opts.onClick,
    }, icon))
  const editor = (kind, label, onSave) => h('input', {
    autoFocus: true, className: S.editor, 'aria-label': label, value: editing.text,
    onChange: (event) => { setEditing({ kind, text: event.currentTarget.value }) },
    onKeyDown: (event) => {
      if (event.key === 'Escape') { setEditing(null); return }
      if (event.key === 'Enter' && !event.nativeEvent.isComposing) { event.preventDefault(); onSave() }
    },
  })
  const editActions = (onSave) => [
    action('save', t('git.save'), Icon('IconCheckOutline16', 'IconCheckOutline16', 14), { disabled: editing.text.trim() === '', onClick: onSave }),
    action('cancel', t('git.cancel'), Icon('IconCloseOutline16', 'IconCloseFill14', 14), { onClick: () => setEditing(null) }),
  ]

  const rows = []
  // 1. Status
  rows.push(h('li', { key: 'status', className: S.row },
    h('span', { className: S.preview }, detail),
    h('div', { className: S.actions },
      [
        action('refresh', t('git.refresh'), Icon('IconRefreshOutline16', 'IconRefreshOutline16', 14), { onClick: () => { load(); if (pr !== void 0) loadPr() } }),
        // G6: the one door that works while the grant is dead. Waking the
        // runner needs GitHub too, so it stays out of the way until then.
        relinkRequired ? relinkAction() : null,
        runnerAsleep && !relinkRequired ? wakeAction() : null,
        sessionSeat ? action('insight-streak', t('git.insight.streak'), Icon('IconChecklistOutline14', 'IconInspectOutline12', 14), { onClick: () => openInsight('streak') }) : null,
        sessionSeat ? action('insight-review', t('git.insight.review'), Icon('IconQueueOutline14', 'IconInspectOutline12', 14), { onClick: () => openInsight('review') }) : null,
        sessionSeat ? action('insight-sessions', t('git.insight.sessions'), Icon('IconBrowseOutline16', 'IconInspectOutline12', 14), { onClick: () => openInsight('sessions') }) : null,
        // Integrate sits with the "{n} behind" it answers. Mid-merge it becomes
        // Finish, so the card offers exactly one next move rather than a button
        // that would start a merge already in progress.
        sessionSeat && status.integrate && status.integrate.integrating
          ? action('integrate-finish', t('git.integrate.finish'), Icon('IconCheckOutline16', 'IconCheckOutline16', 14), {
            onClick: ask('finish', () => run('integrate', async () => { await post('card.integrate.finish', seatArg()); load() })),
          })
          : (sessionSeat && status.integrate && status.integrate.behind > 0
            ? action('integrate', t('git.integrate'), Icon('IconDownloadOutline16', 'IconRefreshOutline16', 14), {
              onClick: ask('integrate', () => run('integrate', async () => { await post('card.integrate', seatArg()); load() })),
            })
            : null),
      ])))
  // 2. Commit — a stage boundary squashes the WIP run above the session
  // base (gw.sessionStageBoundary), so a clean tree with WIP commits is
  // committable. Gating on dirty files alone left Commit dark whenever the
  // watcher had already auto-saved the work (2026-09-02, flow 2 smoke).
  const wipRun = typeof status.wipRun === 'number' ? status.wipRun : 0
  const canCommit = status.health === 'ok' && ((changed !== null && changed > 0) || wipRun > 0)
  const commitPreview = changed === 0 && wipRun > 0 ? t('git.wip', { n: wipRun }) : summary
  const committing = editing !== null && editing.kind === 'commit'
  rows.push(h('li', { key: 'commit', className: S.row },
    committing
      ? editor('commit', t('git.commitSubject'), commit)
      : h('span', { className: S.preview }, canCommit ? commitPreview : t('git.commitHint')),
    h('div', { className: S.actions },
      committing
        ? editActions(commit)
        : [
          canCommit ? action('draft', t('git.askDraft'), Icon('IconSparkle16', 'IconEnhanceOutline16', 14), { onClick: askDraft }) : null,
          action('commit', t('git.commit'), Icon('IconListPenOutline16', 'IconCheckOutline16', 14), { disabled: !canCommit, onClick: () => setEditing({ kind: 'commit', text: '' }) }),
          // D113: Finish sits with Commit because it is the same journey's end
          // — Commit lands the work on main, Finish clears what is left behind.
          // `status.finish` is null on an org seat and on a PARKED session (D40:
          // parked is never deleted), so the button is absent, not merely dark.
          // When present but refused it stays visible and DISABLED, and the
          // preview line says which refusal — a mute dark button teaches nothing.
          status.finish
            ? action('finish', t('git.finish') + (status.finish.can ? '' : ' — ' + (t('git.finish.' + status.finish.reason) || status.finish.reason)),
              Icon('IconArchiveOutline20', 'IconCloseOutline16', 14),
              { disabled: !status.finish.can, onClick: ask('finishSession', finishSession) })
            : null,
        ])))
  // 3. Approve — session seats of a linked org only
  if (status.seat && status.seat.kind === 'session' && status.linked && !status.localOnly) {
    const prAny = pr && pr.ok && pr.pr
    const prMerged = Boolean(prAny && (pr.merged || pr.pr.state === 'merged'))
    const prOpen = Boolean(prAny && !prMerged)
    // G6: while the grant is dead every GitHub read below is a 401, so the
    // row says the one true thing and keeps its place in the card.
    const prText = relinkRequired ? t('git.relinkNeeded')
      : pr === void 0 ? t('git.loading')
      : pr && pr.ok === false ? pr.reason
      : prAny ? t('git.pr.open', { n: pr.pr.number, state: pr.pr.state })
        + (checkState !== 'none' ? ' · ' + t('git.pr.checks', { state: checkState === 'asleep' ? t('git.runner.asleep') : checkState }) : '')
        + (mintChip ? ' · ' + (mintChip.label || String(mintChip)) : '')
      : t('git.pr.none')
    const titling = editing !== null && editing.kind === 'pr'
    rows.push(h('li', { key: 'approve', className: S.row },
      titling
        ? editor('pr', t('git.title.subject'), createPr)
        : h('span', { className: S.preview }, prText),
      h('div', { className: S.actions },
        titling
          ? editActions(createPr)
          : relinkRequired ? [] : [
            action('pr-refresh', t('git.pr.refresh'), Icon('IconRefreshOutline16', 'IconRefreshOutline16', 14), { onClick: loadPr }),
            checkState === 'asleep' ? wakeAction() : null,
            // Q8: the run itself is controllable from the card. All three
            // appear only once a run exists on this branch; re-run waits for
            // it to finish and cancel waits for it to be live, so the pair is
            // never both-enabled on the same run.
            latestRun ? action('ci-rerun', t('git.ci.rerun'), Icon('IconRefreshOutline14', 'IconRefreshOutline16', 14), { disabled: runLive, onClick: rerunCi }) : null,
            latestRun ? action('ci-cancel', t('git.ci.cancel'), Icon('IconStopFill16', 'IconCloseOutline16', 14), { disabled: !runLive, onClick: cancelCi }) : null,
            latestRun && latestRun.url ? action('ci-open', t('git.ci.open'), Icon('IconRightUpOutline16', 'IconBrowseOutline16', 14), { onClick: openRun }) : null,
            // Merge is the one irreversible button: green checks AND an open PR.
            prOpen
              ? action('merge', t('git.pr.merge'), Icon('IconCheckOutline16', 'IconCheckOutline16', 14), { disabled: !(pr.checks && pr.checks.state === 'green' && pr.pr.state === 'open'), onClick: ask('merge', mergePr) })
              : prMerged
                ? action('mint', t('git.mint'), Icon('IconGoalOutline16', 'IconPlusOutline16', 14), { onClick: ask('mint', mint) })
                : action('pr-create', t('git.pr.create'), Icon('IconSendOutline16', 'IconPlusOutline16', 14), { disabled: pr === void 0, onClick: () => setEditing({ kind: 'pr', text: '' }) }),
          ])))
  }

  const confirmModal = confirm === null ? null : h(P.Modal, {
    open: true, onClose: () => setConfirm(null),
    title: confirm.title, description: confirm.body, closeLabel: t('git.cancel'),
    footer: h(React.Fragment, null,
      h(P.Button, { variant: 'outline', autoFocus: true, onClick: () => setConfirm(null) }, t('git.cancel')),
      h(P.Button, { variant: 'outline', onClick: confirmGo }, confirm.ok)),
  })
  const relinkModal = relink === null ? null : h(P.Modal, {
    open: true, onClose: stopRelink,
    title: t('git.relink.title'), description: t('git.relink.desc'), closeLabel: t('git.cancel'),
    footer: h(P.Button, { variant: 'outline', autoFocus: true, onClick: stopRelink }, t('git.cancel')),
  }, h('div', { style: { display: 'flex', flexDirection: 'column', gap: '10px', alignItems: 'flex-start' } },
    h('code', { style: { fontSize: '18px', letterSpacing: '2px' } }, relink.userCode || t('git.relink.starting')),
    relink.verificationUri
      ? h('a', {
        href: relink.verificationUri, target: '_blank', rel: 'noreferrer',
        onClick: (event) => { event.preventDefault(); window.open(relink.verificationUri, '_blank', 'noopener') },
      }, t('git.relink.open'))
      : null))

  return h(React.Fragment, null,
    h('div', { className: S.dock, 'data-git-dock': '' },
    h('div', { className: S.panel },
      h('button', {
        type: 'button', className: S.header, 'aria-controls': listId, 'aria-expanded': expanded,
        'aria-label': expanded ? t('git.collapse') : t('git.expand'),
        disabled: interactionActive,
        onClick: () => { setCollapsed((value) => !value) },
      },
        h('span', { className: S.lead, 'aria-hidden': true }, Icon('IconBranchOutline16', 'IconBranchOutline16', 14)),
        h('span', { className: S.count }, t('git.head', { branch, summary })),
        h('span', { className: S.chevron, 'aria-hidden': true }, expanded ? Icon('IconChevronDownOutline14', 'IconChevronDownOutline14') : Icon('IconChevronUpOutline14', 'IconChevronUpOutline14'))),
      h('ul', { id: listId, className: S.list, hidden: !expanded }, expanded ? rows : null))),
    confirmModal, relinkModal)
}
